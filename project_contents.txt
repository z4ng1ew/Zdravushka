import re
import uuid

from app import config, schedule
from app.models import Booking, ChatResponse, Escalation

DEMO_DOCTORS = [
    {"id": 1, "full_name": "Смирнова Елена Викторовна", "specialty": "Терапевт", "is_active": True},
    {"id": 2, "full_name": "Иванов Пётр Сергеевич", "specialty": "Терапевт", "is_active": True},
    {"id": 4, "full_name": "Лебедев Андрей Игоревич", "specialty": "Офтальмолог", "is_active": True},
]

GREETING = (f"Здравствуйте! Я {config.ASSISTANT_NAME}, виртуальный помощник регистратуры "
            "поликлиники номер семь. Разговор записывается. Чем могу помочь?")

RED_FLAGS = ["боль в груди", "болит грудь", "не могу дышать", "задыха", "потерял сознание",
             "без сознания", "инсульт", "кровотечение"]
WANT_HUMAN = ["оператор", "человек", "сотрудник", "регистратор", "живой"]

SESSIONS: dict[str, dict] = {}


def _summary(s: dict) -> str:
    c = s["collected"]
    parts = [f"{k}: {v}" for k, v in c.items()]
    return "; ".join(parts) or "данных о пациенте нет"


async def handle(session_id: str | None, text: str) -> ChatResponse:
    if not session_id or session_id not in SESSIONS:
        session_id = session_id or uuid.uuid4().hex[:12]
        SESSIONS[session_id] = {"step": "start", "collected": {}, "options": []}
        if not text.strip():
            return ChatResponse(session_id=session_id, reply=GREETING)

    s = SESSIONS[session_id]
    low = text.lower()

    if any(f in low for f in RED_FLAGS):
        s["step"] = "done"
        return ChatResponse(
            session_id=session_id, intent="escalation", done=True, collected=s["collected"],
            reply="Немедленно звоните сто три или сто двенадцать. Соединяю с регистратором.",
            escalation=Escalation(rule="R-01", reason="Угроза жизни", priority="P1",
                                  emergency=True, summary=f"Пациент: «{text}». {_summary(s)}"))
    if any(w in low for w in WANT_HUMAN):
        s["step"] = "done"
        return ChatResponse(
            session_id=session_id, intent="escalation", done=True, collected=s["collected"],
            reply="Соединяю с сотрудником.",
            escalation=Escalation(rule="R-02", reason="Пациент попросил оператора",
                                  priority="P2", summary=_summary(s)))

    step = s["step"]

    if step == "start":
        if any(w in low for w in ["запис", "запиш", "талон", "к врачу"]):
            s["collected"]["специальность"] = next(
                (sp for w, sp in SPECIALTY_WORDS.items() if w in low), "Терапевт")
            s["step"] = "ask_name"
            return ChatResponse(session_id=session_id, intent="booking", collected=s["collected"],
                                reply="Хорошо. Назовите, пожалуйста, фамилию, имя и отчество пациента.")
        return ChatResponse(session_id=session_id, collected=s["collected"],
                            reply="Я могу записать к врачу, перенести или отменить запись. Что нужно?")

    if step == "ask_name":
        s["collected"]["ФИО"] = text.strip().title()
        s["step"] = "ask_birth"
        return ChatResponse(session_id=session_id, intent="booking", collected=s["collected"],
                            reply="Спасибо. Назовите дату рождения.")

    if step == "ask_birth":
        s["collected"]["дата рождения"] = text.strip()
        doctors = [d for d in schedule.demo_doctors() if d["specialty"] == s["collected"]["специальность"]]
        found = schedule.find_free_slots(doctors, set())
        s["options"] = [Booking(doctor_id=d["id"], doctor_name=d["full_name"], specialty=d["specialty"],
                                scheduled_at=schedule.to_api(t), spoken=schedule.say_slot(t))
                        for d, t in found]
        s["step"] = "choose"
        variants = "; ".join(f"{i + 1}: {o.spoken}" for i, o in enumerate(s["options"]))
        return ChatResponse(session_id=session_id, intent="booking", collected=s["collected"],
                            options=s["options"],
                            reply=f"Есть свободное время. {variants}. Какой вариант удобен?")

    if step == "choose":
        m = re.search(r"\d", low)
        words = {"перв": 0, "втор": 1, "трет": 2}
        idx = int(m.group()) - 1 if m else next((v for k, v in words.items() if k in low), None)
        if idx is None or not (0 <= idx < len(s["options"])):
            return ChatResponse(session_id=session_id, intent="booking", collected=s["collected"],
                                options=s["options"], reply="Скажите номер варианта: первый, второй или третий.")
        s["chosen"] = s["options"][idx]
        s["step"] = "confirm"
        o = s["chosen"]
        return ChatResponse(session_id=session_id, intent="booking", collected=s["collected"],
                            booking=o, reply=f"Записываю: врач {o.doctor_name}, {o.spoken}. Верно?")

    if step == "confirm":
        if re.search(r"\b(да|верно|хорошо|подтверждаю)\b", low):
            o = s["chosen"].model_copy(update={"visit_id": 0, "status": "Запланирована"})
            s["step"] = "done"
            doc = schedule.excel_doctor(o.doctor_name) or {}
            room = f", кабинет {doc['cabinet']}" if doc.get("cabinet") else ""
            return ChatResponse(session_id=session_id, intent="booking", collected=s["collected"],
                                booking=o, done=True,
                                reply=f"Готово, вы записаны {o.spoken}{room}. Возьмите паспорт и полис. Всего доброго!")
        s["step"] = "choose"
        return ChatResponse(session_id=session_id, intent="booking", collected=s["collected"],
                            options=s["options"], reply="Хорошо, выберите другой вариант.")

    return ChatResponse(session_id=session_id, done=True, collected=s["collected"],
                        reply="Разговор завершён. Чтобы начать заново, нажмите «Новый звонок».")

import json
from datetime import date, datetime, timedelta, timezone
from functools import lru_cache
from pathlib import Path

from app import config

TZ = timezone(timedelta(hours=config.STAND_TZ_OFFSET_HOURS))

DATE_FMT = "%d.%m.%Y"
DT_FMT = "%d.%m.%Y %H:%M"

SCHEDULE_FILE = Path(__file__).resolve().parent.parent / "data" / "schedule.json"

PRIMARY = {"терапевт", "педиатр", "врач общей практики"}

HOLIDAYS = {date(2026, 11, 4), date(2026, 12, 31), date(2027, 1, 1)}


def now() -> datetime:
    return datetime.now(TZ).replace(tzinfo=None)


def to_api(dt: datetime) -> str:
    return dt.strftime(DT_FMT)


def from_api(s: str) -> datetime:
    return datetime.strptime(s, DT_FMT)


def norm_name(name: str) -> str:
    return " ".join(name.lower().replace("ё", "е").split())


@lru_cache(maxsize=1)
def _data() -> dict:
    raw = json.loads(SCHEDULE_FILE.read_text(encoding="utf-8"))
    doctors = {norm_name(d["full_name"]): d for d in raw["doctors"]}
    free = {norm_name(fio): days for fio, days in raw["free"].items()}
    return {"doctors": doctors, "free": free}


def excel_doctor(full_name: str) -> dict | None:
    return _data()["doctors"].get(norm_name(full_name))


def demo_doctors() -> list[dict]:
    return [{"id": int(d["excel_id"][1:]), "full_name": d["full_name"],
             "specialty": d["specialty"], "is_active": True}
            for d in _data()["doctors"].values()]


def specialties() -> list[str]:
    return sorted({d["specialty"] for d in _data()["doctors"].values()})


def is_primary(specialty: str) -> bool:
    return specialty.strip().lower() in PRIMARY


def day_slots(doctor: dict, d: date) -> list[datetime]:
    days = _data()["free"].get(norm_name(doctor["full_name"]), {})
    times = days.get(d.strftime(DATE_FMT), [])
    return [datetime.strptime(f"{d.strftime(DATE_FMT)} {t}", DT_FMT) for t in times]


def is_valid_slot(doctor: dict, dt: datetime) -> bool:
    return dt > now() and dt in day_slots(doctor, dt.date())


def pgg_deadline(specialty: str, start: datetime) -> datetime:
    if is_primary(specialty):
        return start + timedelta(hours=24)
    d, left = start.date(), 14
    while left > 0:
        d += timedelta(days=1)
        if d.weekday() < 5 and d not in HOLIDAYS:
            left -= 1
    return datetime(d.year, d.month, d.day, 23, 59)


PARTS_OF_DAY = {"утро": (7, 12), "день": (12, 16), "вечер": (16, 21)}


def find_free_slots(doctors: list[dict], busy: set[tuple[int, str]],
                    date_from: date | None = None, date_to: date | None = None,
                    part_of_day: str | None = None, limit: int = 3,
                    search_days: int = 30) -> list[tuple[dict, datetime]]:
    start = now() + timedelta(minutes=30)
    day = max(date_from or start.date(), start.date())
    last = date_to or (day + timedelta(days=search_days))
    hours = PARTS_OF_DAY.get(part_of_day) if part_of_day else None

    found = []
    while day <= last and len(found) < limit:
        candidates = []
        for doc in doctors:
            for slot in day_slots(doc, day):
                if slot < start:
                    continue
                if hours and not (hours[0] <= slot.hour < hours[1]):
                    continue
                if (doc["id"], to_api(slot)) in busy:
                    continue
                candidates.append((slot, doc))
                break
        for slot, doc in sorted(candidates, key=lambda x: x[0]):
            if len(found) < limit:
                found.append((doc, slot))
        day += timedelta(days=1)
    return found


def weekly_text(full_name: str) -> str | None:
    doc = excel_doctor(full_name)
    if not doc or not doc["weekly"]:
        return None
    return ", ".join(f"{d} {h}" for d, h in doc["weekly"].items())


WEEKDAYS_ACC = ["в понедельник", "во вторник", "в среду", "в четверг",
                "в пятницу", "в субботу", "в воскресенье"]
MONTHS_GEN = ["января", "февраля", "марта", "апреля", "мая", "июня", "июля",
              "августа", "сентября", "октября", "ноября", "декабря"]
DAY_ORDINALS = ["", "первого", "второго", "третьего", "четвёртого", "пятого", "шестого",
                "седьмого", "восьмого", "девятого", "десятого", "одиннадцатого",
                "двенадцатого", "тринадцатого", "четырнадцатого", "пятнадцатого",
                "шестнадцатого", "семнадцатого", "восемнадцатого", "девятнадцатого",
                "двадцатого", "двадцать первого", "двадцать второго", "двадцать третьего",
                "двадцать четвёртого", "двадцать пятого", "двадцать шестого",
                "двадцать седьмого", "двадцать восьмого", "двадцать девятого",
                "тридцатого", "тридцать первого"]
UNITS = ["ноль", "один", "два", "три", "четыре", "пять", "шесть", "семь", "восемь",
         "девять", "десять", "одиннадцать", "двенадцать", "тринадцать", "четырнадцать",
         "пятнадцать", "шестнадцать", "семнадцать", "восемнадцать", "девятнадцать"]
TENS = {2: "двадцать", 3: "тридцать", 4: "сорок", 5: "пятьдесят"}


def number_words(n: int) -> str:
    if n < 20:
        return UNITS[n]
    tens, units = divmod(n, 10)
    return TENS[tens] + (f" {UNITS[units]}" if units else "")


def say_time(dt: datetime) -> str:
    h = number_words(dt.hour)
    if dt.minute == 0:
        return f"{h} ноль-ноль"
    if dt.minute < 10:
        return f"{h} ноль {number_words(dt.minute)}"
    return f"{h} {number_words(dt.minute)}"


def say_date(d: date) -> str:
    base = f"{WEEKDAYS_ACC[d.weekday()]}, {DAY_ORDINALS[d.day]} {MONTHS_GEN[d.month - 1]}"
    today = now().date()
    if d == today:
        return f"сегодня, {base}"
    if d == today + timedelta(days=1):
        return f"завтра, {base}"
    return base


def say_slot(dt: datetime) -> str:
    return f"{say_date(dt.date())}, в {say_time(dt)}"

===== app/stand_client.py =====
import asyncio
import time
from collections import deque

import httpx

from app import config


class StandError(Exception):
    def __init__(self, status: int, code: str, message: str):
        super().__init__(f"{status} {code}: {message}")
        self.status = status
        self.code = code
        self.message = message


class RateLimiter:
    def __init__(self, max_calls: int, period: float = 60.0):
        self.max_calls = max_calls
        self.period = period
        self.calls = deque()
        self.lock = asyncio.Lock()

    async def wait(self):
        async with self.lock:
            while True:
                now = time.monotonic()
                while self.calls and now - self.calls[0] >= self.period:
                    self.calls.popleft()
                if len(self.calls) < self.max_calls:
                    self.calls.append(now)
                    return
                await asyncio.sleep(self.period - (now - self.calls[0]) + 0.05)


class StandClient:
    def __init__(self, base_url: str = config.STAND_URL, token: str = config.STAND_TOKEN,
                 transport: httpx.AsyncBaseTransport | None = None):
        self.http = httpx.AsyncClient(
            base_url=base_url,
            headers={"Authorization": f"Bearer {token}"},
            timeout=config.STAND_TIMEOUT,
            transport=transport,
        )
        self.limiter = RateLimiter(config.STAND_RATE_LIMIT)
        self._doctors: list[dict] | None = None

    async def close(self):
        await self.http.aclose()

    async def _request(self, method: str, path: str, **kwargs):
        for attempt in range(3):
            await self.limiter.wait()
            resp = await self.http.request(method, path, **kwargs)

            if resp.status_code == 429:
                delay = float(resp.headers.get("Retry-After", "10"))
                await asyncio.sleep(delay)
                continue

            if resp.status_code >= 400:
                try:
                    body = resp.json()
                except ValueError:
                    body = {}
                raise StandError(resp.status_code,
                                 body.get("code", "НЕИЗВЕСТНО"),
                                 body.get("message", resp.text))
            return resp.json()

        raise StandError(429, "ПРЕВЫШЕН_ЛИМИТ", "Стенд трижды ответил 429")

    async def _get_all(self, path: str, params: dict | None = None) -> list[dict]:
        params = dict(params or {})
        result, offset = [], 0
        while True:
            page = await self._request("GET", path, params={**params, "limit": 50, "offset": offset})
            items = page if isinstance(page, list) else page.get("items", [])
            result.extend(items)
            if len(items) < 50:
                return result
            offset += 50

    async def doctors(self, refresh: bool = False) -> list[dict]:
        if self._doctors is None or refresh:
            self._doctors = await self._get_all("/doctors")
        return self._doctors

    async def active_doctors(self, specialty: str | None = None) -> list[dict]:
        result = [d for d in await self.doctors() if d["is_active"]]
        if specialty:
            s = specialty.strip().lower()
            result = [d for d in result if d["specialty"].lower() == s]
        return result

    async def find_client(self, last_name: str, first_name: str, birth_date: str) -> dict | None:
        found = await self._get_all("/clients", {
            "last_name": last_name.strip().title(),
            "first_name": first_name.strip().title(),
            "birth_date": birth_date,
        })
        return found[0] if found else None

    async def create_client(self, last_name: str, first_name: str, birth_date: str,
                            middle_name: str | None = None, comment: str | None = None) -> dict:
        body = {"last_name": last_name.strip().title(),
                "first_name": first_name.strip().title(),
                "birth_date": birth_date}
        if middle_name:
            body["middle_name"] = middle_name.strip().title()
        if comment:
            body["comment"] = comment
        return await self._request("POST", "/clients", json=body)

    async def update_client_comment(self, client_id: int, comment: str) -> dict:
        return await self._request("PATCH", f"/clients/{client_id}", json={"comment": comment})

    async def create_visit(self, client_id: int, doctor_id: int, scheduled_at: str) -> dict:
        return await self._request("POST", "/visits", json={
            "client_id": client_id, "doctor_id": doctor_id, "scheduled_at": scheduled_at})

    async def list_visits(self, client_id: int | None = None, doctor_id: int | None = None,
                          status: str | None = None, date_from: str | None = None,
                          date_to: str | None = None) -> list[dict]:
        params = {k: v for k, v in {
            "client_id": client_id, "doctor_id": doctor_id, "status": status,
            "date_from": date_from, "date_to": date_to}.items() if v is not None}
        return await self._get_all("/visits", params)

    async def reschedule_visit(self, visit_id: int, scheduled_at: str,
                               doctor_id: int | None = None) -> dict:
        body = {"scheduled_at": scheduled_at}
        if doctor_id is not None:
            body["doctor_id"] = doctor_id
        return await self._request("PATCH", f"/visits/{visit_id}", json=body)

    async def cancel_visit(self, visit_id: int) -> dict:
        return await self._request("PATCH", f"/visits/{visit_id}", json={"status": "Отменена"})

    async def is_slot_taken(self, doctor_id: int, scheduled_at: str) -> bool:
        day = scheduled_at.split(" ")[0]
        visits = await self.list_visits(doctor_id=doctor_id, status="Запланирована",
                                        date_from=day, date_to=day)
        return any(v["scheduled_at"] == scheduled_at for v in visits)

import asyncio

from app import config
from app.stand_client import StandClient, StandError


async def main():
    if not config.STAND_TOKEN:
        print("Нет токена: впишите STAND_TOKEN в файл .env")
        return

    client = StandClient()
    try:
        doctors = await client.doctors()
        print(f"Врачей в справочнике: {len(doctors)}")
        by_specialty = {}
        for d in doctors:
            by_specialty.setdefault(d["specialty"], []).append(d)
        for specialty, items in sorted(by_specialty.items()):
            active = sum(1 for d in items if d["is_active"])
            print(f"  {specialty}: {len(items)} (активных {active})")

        visits = await client.list_visits()
        print(f"Записей у нашей команды: {len(visits)}")
    except StandError as e:
        print(f"Ошибка стенда: {e.status} {e.code} — {e.message}")
    finally:
        await client.close()


if __name__ == "__main__":
    asyncio.run(main())
    

===== scripts/build_schedule.py =====
import json
import sys
from datetime import datetime
from pathlib import Path

import openpyxl

OUT = Path("data/schedule.json")


def hhmm(t) -> str:
    return t.strftime("%H:%M") if hasattr(t, "strftime") else str(t)[:5]


def main(xlsx_path: str):
    print(f"Читаю {xlsx_path} ... (около минуты)")
    wb = openpyxl.load_workbook(xlsx_path, read_only=True)

    today = None
    for row in wb["Описание"].iter_rows(values_only=True):
        if row[0] == "Сегодня":
            today = row[1]
    today = today or datetime.now()

    doctors = {}
    for r in wb["Врачи"].iter_rows(min_row=2, values_only=True):
        if not r[0]:
            continue
        doctors[r[1]] = {
            "excel_id": r[0], "full_name": r[1], "department": r[2], "specialty": r[3],
            "patients": r[4], "cabinet": str(r[5]), "slot_min": int(r[6]), "schedule": r[7],
            "weekly": {},
        }

    days = ["Пн", "Вт", "Ср", "Чт", "Пт", "Сб", "Вс"]
    for r in wb["График"].iter_rows(min_row=2, values_only=True):
        if r[1] in doctors:
            doctors[r[1]]["weekly"] = {d: v for d, v in zip(days, r[4:11]) if v and v != "—"}

    free: dict[str, dict[str, list[str]]] = {}
    count = 0
    for r in wb["Слоты"].iter_rows(min_row=2, values_only=True):
        day, fio, status = r[1], r[4], r[11]
        if not day or day < today or status != "Свободен":
            continue
        free.setdefault(fio, {}).setdefault(day.strftime("%d.%m.%Y"), []).append(hhmm(r[8]))
        count += 1

    OUT.parent.mkdir(parents=True, exist_ok=True)
    data = {"source": Path(xlsx_path).name, "today": today.strftime("%d.%m.%Y"),
            "doctors": list(doctors.values()), "free": free}
    OUT.write_text(json.dumps(data, ensure_ascii=False), encoding="utf-8")
    print(f"Готово: {len(doctors)} врачей, {count} свободных слотов → {OUT} "
          f"({OUT.stat().st_size // 1024} КБ)")


if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Использование: python -m scripts.build_schedule путь/к/расписанию.xlsx")
        sys.exit(1)
    main(sys.argv[1])
