  const lines = feed.filter((f): f is Extract<FeedItem, { type: "line" }> => f.type === "line").map((f) => f.line);

  const say = useCallback((speaker: LiveLine["speaker"], text: string, confidence?: number) => {
    setFeed((f) => [...f, { type: "line", line: { speaker, text, at: nowHM(), confidence } }]);
  }, []);

  useEffect(() => { phaseRef.current = phase; }, [phase]);

  function showError(e: unknown) {
    const err = e instanceof ApiError ? e : new ApiError(null, "?", "?", String(e));
    setFeed((f) => [...f, { type: "error", status: err.status, detail: err.message, at: nowHM() }]);
  }

  function changeMode(m: CallMode) {
    modeRef.current = m;
    setMode(m);
  }

  // ---- проверка сервера при открытии
  useEffect(() => {
    health().then(() => setOnline(true)).catch((e) => { setOnline(false); showError(e); });
  }, []);

  // ---- связь с кабинетом регистратора
  useEffect(() => {
    live.current = openLive((m) => {
      if (m.type === "hello" && snapshot.current) live.current?.send({ type: "state", call: snapshot.current });
      if (m.type === "takeover") {
        recRef.current?.cancel(); recRef.current = null; setListening(false);
        setPhase("active");
        setOperator(m.operator);
        changeMode("operator");
      }
      if (m.type === "release") {
        setOperator(null);
        changeMode("ai");
        say("bot", "Я снова с вами. Чем ещё могу помочь?");
      }
      if (m.type === "operator_say") say("operator", m.text);
    });
    return () => live.current?.close();
  }, [say]);

  useEffect(() => {
    const call: LiveCall = { phase, mode, seconds, intent, fields, lines, operator };
    snapshot.current = call;
    live.current?.send({ type: "state", call });
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [phase, mode, seconds, intent, fields, feed, operator]);

  useEffect(() => {
    if (phase !== "active") return;
    const t = setInterval(() => setSeconds((s) => s + 1), 1000);
    return () => clearInterval(t);
  }, [phase]);

  useEffect(() => {
    chatRef.current?.scrollTo({ top: chatRef.current.scrollHeight, behavior: "smooth" });
  }, [feed]);

  // ---- разговор с бэкендом
  function clearAll() {
    setFeed([]); setIntent(null); setFields({}); setBooking(null); setSeconds(0);
    setOptions([]); setEscalation(null); setOperator(null); changeMode("ai");
    sessionRef.current = null;
    recRef.current?.cancel(); recRef.current = null; setListening(false);
  }

  function applyChat(r: ChatResponse) {
    sessionRef.current = r.session_id || sessionRef.current;
    if (r.reply) say("bot", r.reply);
    if (r.intent) setIntent(INTENT_LABEL[r.intent] ?? r.intent);
    setFields(r.collected);
    setOptions(r.options);
    if (r.booking) setBooking(toBooking(r.booking, r.intent));
    if (r.escalation) {
      // бот передаёт пациента человеку — звонок не обрываем, ждём регистратора
      setEscalation(r.escalation);
      setBooking({ kind: "handoff", title: "Передано регистратору", note: `${r.escalation.reason}. ${r.escalation.summary}` });
      changeMode("waiting");
      return;
    }
    if (r.done) setPhase("ended");
  }

  async function send(text: string): Promise<ChatResponse | null> {
    setSending(true);
    try {
      const r = await chat(sessionRef.current, text);
      applyChat(r);
      setOnline(true);
      return r;
    } catch (e) {
      showError(e);
      if (!(e instanceof ApiError) || e.status === null || e.status >= 500) setOnline(false);
      return null;
    } finally {
      setSending(false);
    }
  }

  async function startCall(firstPhrase?: string) {
    clearAll();
    setPhase("active");
    phaseRef.current = "active";
    const r = await send(""); // приветствие и session_id
    if (!r) { setPhase("ended"); voiceRef.current = false; return; }
    if (firstPhrase) { say("patient", firstPhrase); await afterBotTurn(await send(firstPhrase)); return; }
    await afterBotTurn(r);
  }

  function endCall() {
    recRef.current?.cancel(); recRef.current = null;
    setListening(false);
    voiceRef.current = false;
    setPhase("ended");
  }

  // ---- голос
  /** Озвучить ответ бота (если на бэкенде есть /api/tts). Ошибки — только в журнал. */
  async function speakText(text: string) {
    if (!text || ttsOk.current === false) return;
    try {
      const audio = await tts(text);
      ttsOk.current = true;
      setSpeaking(true);
      await play(audio);
    } catch (e) {
      if (e instanceof ApiError && e.status !== null && [404, 405, 501].includes(e.status)) ttsOk.current = false;
    } finally {
      setSpeaking(false);
    }
  }

  /** После ответа бота в голосовом звонке: озвучить и снова слушать пациента. */
  async function afterBotTurn(r: ChatResponse | null) {
    if (!r || !voiceRef.current) return;
    await speakText(r.reply);
    if (!r.done && !r.escalation && phaseRef.current === "active" && modeRef.current === "ai") startListening();
  }

  async function startListening() {
    if (recRef.current) return;
    try {
      recRef.current = await startRecording(() => finishListening(false));
      setListening(true);
    } catch (err) {
      voiceRef.current = false;
      const e = new ApiError(null, "MIC", "микрофон", `Нет доступа к микрофону: ${err instanceof Error ? err.message : err}`);
      logError(e, "голос");
      showError(e);
    }
  }

  /** manual — пациент сам нажал «отправить»: тогда отправляем запись, даже если она тихая */
  async function finishListening(manual = false) {
    const rec = recRef.current;
    if (!rec) return;
    recRef.current = null;
    setListening(false);
    const heard = rec.heardSpeech();
    const audio = await rec.stop();
    if (phaseRef.current !== "active" || modeRef.current !== "ai") return;
    if (!heard && !manual) { startListening(); return; } // молчание — слушаем дальше
    setTranscribing(true);
    try {
      const res = await asr(audio);
      if (!res.text) { startListening(); return; }
      say("patient", res.text, res.confidence ?? undefined);
      setOptions([]);
      await afterBotTurn(await send(res.text));
    } catch (e) {
      voiceRef.current = false; // не крутим ошибку по кругу — пациент нажмёт микрофон ещё раз
      showError(e);
    } finally {
      setTranscribing(false);
    }
  }

  function callRegistrar() {
    if (phase !== "active") setPhase("active");
    changeMode("waiting");
    say("bot", "Соединяю вас с регистратором, оставайтесь на линии.");
  }

  function chooseOption(i: number) {
    if (sending) return;
    const o = options[i];
    say("patient", o?.spoken ? `${i + 1} — ${o.spoken}` : String(i + 1));
    setOptions([]);
    recRef.current?.cancel(); recRef.current = null; setListening(false);
    send(String(i + 1)).then(afterBotTurn);
  }

  function onSubmit(e: FormEvent) {
    e.preventDefault();
    const text = input.trim();
    if (!text || sending) return;
    setInput("");
    if (mode !== "ai") { say("patient", text); return; } // с регистратором — просто передаём фразу
    if (phase !== "active") { startCall(text); return; }
    say("patient", text);
    setOptions([]);
    send(text);
  }

  function pressMic() {
    if (sending || transcribing || speaking) return;
    const id = Date.now();
    setBursts((b) => [...b, id]);
    setTimeout(() => setBursts((b) => b.filter((x) => x !== id)), 700);
    if (phase !== "active") { voiceRef.current = true; startCall(); return; }
    if (listening) { finishListening(true); return; } // нажал ещё раз — отправить сразу
    if (mode === "ai") { voiceRef.current = true; startListening(); }
  }

  const last = lines[lines.length - 1];
  const status =
    phase === "idle" ? "Здравушка готов помочь"
    : phase === "ended" ? "Звонок завершён"
    : mode === "waiting" ? "Соединяю с регистратором…"
    : mode === "operator" ? (last?.speaker === "operator" ? "Говорит регистратор…" : "Регистратор слушает вас…")
    : listening ? "Слушаю — говорите"
    : transcribing ? "Распознаю речь…"
    : sending ? "Здравушка думает…"
    : speaking ? "Здравушка говорит…"
    : !last || last.speaker === "bot" ? "Здравушка говорит…" : "Слушаю вас…";

  const pill =
    phase !== "active" ? { cls: "pill-ghost", text: phase === "ended" ? "Звонок завершён" : "Нет звонка" }
    : mode === "operator" ? { cls: "pill-green", text: `На линии регистратор${operator ? " " + operator : ""}` }
    : mode === "waiting" ? { cls: "pill-orange", text: "Ожидание регистратора" }
    : { cls: "pill-blue", text: "Ведёт ИИ" };

  const canCall = phase !== "active" && !sending;
  const canTalk = phase === "active" && mode === "ai" && !sending && !transcribing && !speaking;

  const recordStatus =
    booking && booking.kind !== "handoff" ? { cls: booking.kind === "booked" ? "tag-ok" : booking.kind === "cancelled" ? "" : "tag-warn", text: booking.title }
    : mode !== "ai" || booking?.kind === "handoff" ? { cls: "tag-warn", text: mode === "operator" ? "Ведёт регистратор" : "Передано регистратору" }
    : { cls: "", text: "Записи пока нет" };
  const summaryKey = `${intent}|${Object.keys(fields).length}|${recordStatus.text}`;
  const hasNews = !detailsOpen && summaryKey !== seen && (intent !== null || booking !== null || mode !== "ai");

  function toggleDetails() {
    setDetailsOpen((o) => !o);
    setSeen(summaryKey);
  }

  return (
    <div className="page">
      <header className="topbar">
        <Brand />
        <div className="topbar-right">
          <span className={`live ${phase === "active" ? "on" : ""}`}
            data-tip={phase === "active" ? "Идёт разговор" : "Сейчас никто не звонит"} data-tip-pos="bottom">
            <i />{phase === "active" ? "На линии" : "Свободно"}
          </span>
          <span className="timer" data-tip="Длительность звонка" data-tip-pos="bottom">{hms(seconds)}</span>
          <span className={`pill ${pill.cls}`}
            data-tip="Кто сейчас ведёт разговор: ИИ или живой регистратор" data-tip-pos="bottom">{pill.text}</span>
          <Link href="/staff/live" className="link-staff"
            data-tip="Кабинет регистратуры: живые звонки, очередь, журнал, напоминания" data-tip-pos="bottom">
            Для сотрудников →
          </Link>
        </div>
      </header>

      <section className="panel call-panel">
        <div className="call-left">
          <div className="greeting" data-tip="Здравушка — голосовой помощник. Записывает, переносит и отменяет приёмы, отвечает на вопросы о работе поликлиники.">
            {/* eslint-disable-next-line @next/next/no-img-element */}
            <img src="/img/mascot-circle.png" width={118} height={118} alt="Здравушка" className="mascot-circle" />
            <p>Привет! Я Здравушка. Чем могу помочь?</p>
          </div>

          <div className={`mic-wrap ${canCall || (canTalk && !listening) ? "mic-idle" : ""} ${listening ? "mic-listen" : ""}`}>

                      <span className="ok-ico"><CheckCircle /></span>
                      <div><small>{k}</small><div>{v}</div></div>
                    </li>
                  ))}
                </ul>
              </div>
            </div>

            <div>
              {booking && booking.kind !== "handoff" ? (
                <div className="record">
                  <div className="doctor-head">
                    <Mascot size={86} />
                    <div className="doctor-name">
                      <b>{booking.doctor}</b>
                      <span>{booking.specialty}</span>
                    </div>
                    <div className="doctor-photo">
                      {(booking.doctor ?? "").split(" ").slice(0, 2).map((w) => w[0]).join("")}
                    </div>
                  </div>
                  <div className="record-grid">
                    <div><CalendarIcon /><small>Дата</small><div>{booking.date}</div></div>
                    <div><ClockIcon /><small>Время</small><div>{booking.time}</div></div>
                  </div>
                  <div className={`result ${booking.kind === "pending" ? "result-warn" : booking.kind === "cancelled" ? "result-off" : "result-ok"}`}>
                    <span className="ok-ico">{booking.kind === "booked" ? <CheckCircle size={22} /> : <ClockIcon size={22} />}</span>
                    <div><b>{booking.title}</b><div>{booking.note}</div></div>
                  </div>
                </div>
              ) : mode !== "ai" || booking?.kind === "handoff" ? (
                <div className="record">
                  <div className={`result ${mode === "operator" ? "result-op" : "result-warn"}`}>
                    <span className="ok-ico"><HeadsetIcon size={22} /></span>
                    <div>
                      <b>{mode === "operator" ? "Разговор ведёт регистратор" : "Передано регистратору"}</b>
                      <div>{booking?.kind === "handoff" ? booking.note : "Регистратор видит весь разговор и то, что понял ИИ."}</div>
                    </div>
                  </div>
                </div>
              ) : (
                <div className="record record-empty">
                  <Mascot size={80} />
                  <p>Здесь появится запись, как только Здравушка оформит её в МИС.</p>
                </div>
              )}
            </div>
          </div>
        )}
      </section>
    </div>
  );
}

        <select value={filter} onChange={(e) => setFilter(e.target.value as CallResult | "все")}>
          <option value="все">все</option>
          {results.map((r) => <option key={r} value={r}>{r}</option>)}
        </select>
      </div>

      <div className="card" style={{ padding: 0 }}>
        <table>
          <thead>
            <tr>
              <th>Время</th>
              <th>Тип</th>
              <th>Цель обращения</th>
              <th>Пациент</th>
              <th>Длит.</th>
              <th>Итог</th>
            </tr>
          </thead>
          <tbody>
            {shown.map((c) => (

import { getHandoffs, updateHandoff } from "@/lib/api";

import { ApiError } from "@/lib/errors";

const order: Record<HandoffStatus, number> = { новая: 0, "в работе": 1, перезвонили: 2, закрыта: 3 };

export default function OperatorPage() {
  const [items, setItems] = useState<Handoff[]>([]);
  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [error, setError] = useState<ApiError | null>(null);

  useEffect(() => {
    getHandoffs().catch((e: ApiError) => { setError(e); return [] as Handoff[]; }).then((list) => {
      setItems(list);
      setSelectedId(list[0]?.id ?? null);
    });
  }, []);

  // Срочные и новые — наверх
  const sorted = useMemo(
    () =>
      [...items].sort(
        (a, b) => Number(b.urgent) - Number(a.urgent) || order[a.status] - order[b.status],
      ),
    [items],
  );

  const openCount = items.filter((h) => h.status === "новая" || h.status === "в работе").length;

const byTime = (l: Reminder[]) => [...l].sort((a, b) => a.scheduled_at.localeCompare(b.scheduled_at));

export default function RemindersPage() {
  const [items, setItems] = useState<Reminder[]>([]);
  const [running, setRunning] = useState(false);
  const [error, setError] = useState<ApiError | null>(null);
  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);

  useEffect(() => {
    getReminders().then((l) => setItems(byTime(l))).catch((e: ApiError) => setError(e));
    return () => timers.current.forEach(clearTimeout);
  }, []);

  async function run() {
    setRunning(true);
    setError(null);
    try {
      await runReminders();
      // бэкенд обзванивает сам — перечитываем статусы через несколько секунд
      timers.current.push(setTimeout(async () => {
        try { setItems(byTime(await getReminders())); } catch (e) { setError(e as ApiError); }
        setRunning(false);
      }, 5000));
    } catch (e) {
      setError(e as ApiError);
      setRunning(false);
    }
  }

  const count = (s: ReminderStatus) => items.filter((r) => r.status === s).length;

  return (
    <>
      <h1 className="page-title">Напоминания</h1>
      <p className="page-sub">
        За день до визита ассистент звонит пациенту, получает подтверждение, а при отказе предлагает перенос
        или отменяет запись — результат сразу записывается в МИС.
      </p>
      {error && <ErrorBlock status={error.status} detail={error.message} />}

      <div className="grid grid-5">
        {summaryOrder.map((s) => (
          <div className="card" key={s}>
            <div className="kpi-label">{s}</div>
            <div className="kpi-value">{count(s)}</div>
          </div>
        ))}
      </div>

      <div className="card mt" style={{ padding: 0 }}>
        <div className="btn-row" style={{ justifyContent: "space-between", alignItems: "center", padding: "14px 16px" }}>
          <b>Приёмы на 15 октября</b>
          <button className="btn btn-primary" onClick={run} disabled={running || count("ожидает") === 0}>
            {running ? "Идёт обзвон…" : "Запустить обзвон"}
          </button>
        </div>
        <table>
          <thead>
            <tr>
              <th>Время приёма</th>
              <th>Пациент</th>
              <th>Врач</th>
              <th>Статус</th>
            </tr>
          </thead>
          <tbody>
            {items.map((r) => (
              <tr key={r.visit_id}>
                <td>{r.scheduled_at.split(" ")[1]}</td>
                <td>{r.patient}</td>
                <td>{r.doctor} <span className="muted small">· {r.specialty}</span></td>
                <td>
                  <ReminderTag value={r.status} />
                  {r.new_time && <div className="small muted">новое время: {r.new_time}</div>}
                  {r.status === "не ответил" && <div className="small muted">повторный звонок через 2 часа</div>}
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </>
  );
}

// Экономический эффект: простые прозрачные формулы, чтобы их можно было проговорить на защите
export function economics() {
  const visitPrice = 1200; // ₽, средняя стоимость приёма по ОМС (из задания)
  const noShowCut = 0.35; // напоминания снижают неявки на 30–40% (берём середину)
  const noShowsPerDay = BASELINE.visitsPerDay * BASELINE.noShowShare;
  const savedVisitsPerDay = noShowsPerDay * noShowCut;
  const rubPerDay = savedVisitsPerDay * visitPrice;

  const answeredPerDay = BASELINE.callsPerDay * (1 - BASELINE.missedShare);
  const automatedPerDay = answeredPerDay * TARGETS.automatedShare;
  const hoursFreedPerDay = (automatedPerDay * BASELINE.avgTalkSec) / 3600;

  return {
    noShowsPerDay,
    savedVisitsPerDay,
    rubPerDay,
    rubPerMonth: rubPerDay * 21, // рабочих дней
    automatedPerDay,
    hoursFreedPerDay,
  };
}

export function describe(status: number | null): string {
  if (status === null) return "Нет связи с сервером";
  return STATUS_TEXT[status] ?? (status >= 500 ? "Ошибка сервера" : "Ошибка запроса");
}

export function readErrors(): ErrorEntry[] {
  try {
    return JSON.parse(localStorage.getItem(KEY) ?? "[]");
  } catch {
    return [];
  }
}

export function clearErrors(): void {
  try { localStorage.removeItem(KEY); } catch { /* хранилище недоступно */ }
}

export function logError(e: ApiError, where: string): void {
  const entry: ErrorEntry = {
    at: new Date().toISOString(), status: e.status, method: e.method, path: e.path, message: e.message, where,
  };
  console.error(`[${where}] ${e.status ?? "нет связи"} ${e.method} ${e.path}: ${e.message}`);
  try {
    const list = [entry, ...readErrors()].slice(0, MAX);
    localStorage.setItem(KEY, JSON.stringify(list));
  } catch { /* хранилище недоступно — остаётся консоль */ }
}

/** Текст лога для скачивания: одна строка — одна ошибка. */
export function toLogText(list: ErrorEntry[]): string {
  const head = "# ClinicAI — журнал ошибок\n# время | код | запрос | где | описание\n";
  return head + list.map((e) =>
    [e.at, e.status ?? "---", `${e.method} ${e.path}`, e.where, e.message.replace(/\s+/g, " ")].join(" | "),
  ).join("\n") + "\n";
}

const API = (process.env.NEXT_PUBLIC_API_URL ?? "").replace(/\/$/, "");

/** Запрос к бэкенду. Любая ошибка записывается в журнал и выбрасывается как ApiError.
 *  body: объект → JSON, FormData → файл (multipart). as: "json" (по умолчанию) или "blob" (звук). */
export async function request<T>(
  method: string, path: string, where: string, body?: unknown, as: "json" | "blob" = "json",
): Promise<T> {
  const isForm = typeof FormData !== "undefined" && body instanceof FormData;
  let r: Response;
  try {
    r = await fetch(`${API}${path}`, {
      method,
      cache: "no-store",
      headers: body !== undefined && !isForm ? { "Content-Type": "application/json" } : undefined,
      body: body === undefined ? undefined : isForm ? (body as FormData) : JSON.stringify(body),
    });
  } catch (err) {
    const e = new ApiError(null, method, path, err instanceof Error ? err.message : "сеть недоступна");
    logError(e, where);
    throw e;
  }
  if (!r.ok) {
    let detail = r.statusText;
    try {
      const j = await r.json();
      detail = typeof j.detail === "string" ? j.detail : j.message ?? JSON.stringify(j.detail ?? j);
    } catch { /* тело не JSON */ }
    const e = new ApiError(r.status, method, path, detail || describe(r.status));
    logError(e, where);
    throw e;
  }
  return (as === "blob" ? r.blob() : r.json()) as Promise<T>;
}

// «Живой» звонок между экраном пациента и кабинетом регистратора.

// одного сайта (без сервера). На бэкенде этот же обмен пойдёт через WebSocket:
// формат сообщений тот же.

import type { Speaker } from "./types";

export type CallMode = "ai" | "waiting" | "operator"; // ведёт ИИ / ждёт регистратора / ведёт регистратор

export interface LiveLine {
  speaker: Speaker;
  text: string;
  at: string;
  confidence?: number; // для реплик пациента: уверенность распознавания
}

export interface LiveCall {

  seconds: number;
  intent: string | null;
  fields: Record<string, string>;
  lines: LiveLine[];
  operator: string | null;

export type LiveMessage =

  | { type: "hello" } // регистратор открыл вкладку и просит текущее состояние
  | { type: "takeover"; operator: string } // регистратор перехватил разговор
  | { type: "release" } // регистратор вернул разговор ИИ
  | { type: "operator_say"; text: string }; // реплика регистратора

const NAME = "clinicai-live";

export function openLive(onMessage: (m: LiveMessage) => void): { send: (m: LiveMessage) => void; close: () => void } {
  if (typeof window === "undefined" || !("BroadcastChannel" in window)) {
    return { send: () => {}, close: () => {} };
  }
  const ch = new BroadcastChannel(NAME);
  ch.onmessage = (e) => onMessage(e.data as LiveMessage);
  return { send: (m) => ch.postMessage(m), close: () => ch.close() };
}

// Демо-оценка уверенности распознавания (на бэкенде её отдаёт ASR)
export function fakeConfidence(text: string): number {
  let h = 0;
  for (const c of text) h = (h * 31 + c.charCodeAt(0)) % 1000;
  return 0.88 + (h % 11) / 100; // 0,88–0,98
}

// Голос: запись с микрофона, распознавание (/api/asr → Whisper) и озвучка ответов (/api/tts).
// Микрофон в браузере работает только на https:// или http://localhost.

import { request } from "./errors";

const WHERE = "голос";

export interface AsrResult {
  text: string;
  confidence: number | null; // 0–1, если распознаватель её отдаёт
}

/** Отправить запись на распознавание. Бэкенд проксирует её в Whisper. */
export async function asr(audio: Blob): Promise<AsrResult> {
  const form = new FormData();
  const ext = audio.type.includes("ogg") ? "ogg" : audio.type.includes("mp4") ? "m4a" : "webm";
  form.append("file", audio, `speech.${ext}`);
  const r = await request<Partial<AsrResult>>("POST", "/api/asr", WHERE, form);
  return { text: (r.text ?? "").trim(), confidence: typeof r.confidence === "number" ? r.confidence : null };
}

/** Озвучить ответ. Возвращает звук или выбрасывает ApiError (например, 404 — озвучки ещё нет). */
export async function tts(text: string): Promise<Blob> {
  return request<Blob>("POST", "/api/tts", WHERE, { text }, "blob");
}

export interface Recording {
  stop: () => Promise<Blob>; // закончить и получить звук
  cancel: () => void; // выбросить запись
  heardSpeech: () => boolean; // был ли голос (а не только тишина)
}

/**
 * Начать запись. Сама остановится (onAutoStop), когда человек замолчит после фразы
 * или пройдёт maxMs. Тишину определяем по громкости сигнала.
 */
export async function startRecording(
  onAutoStop: () => void,
  { silenceMs = 1300, maxMs = 15000, threshold = 0.025 } = {},
): Promise<Recording> {
  const stream = await navigator.mediaDevices.getUserMedia({
    audio: { echoCancellation: true, noiseSuppression: true, channelCount: 1 },
  });
  const mime = ["audio/webm;codecs=opus", "audio/webm", "audio/ogg;codecs=opus", "audio/mp4"]
    .find((m) => typeof MediaRecorder !== "undefined" && MediaRecorder.isTypeSupported(m));
  const rec = new MediaRecorder(stream, mime ? { mimeType: mime } : undefined);
  const chunks: BlobPart[] = [];
  rec.ondataavailable = (e) => { if (e.data.size) chunks.push(e.data); };

  // простой детектор паузы
  const ctx = new AudioContext();
  const analyser = ctx.createAnalyser();
  analyser.fftSize = 1024;
  ctx.createMediaStreamSource(stream).connect(analyser);
  const buf = new Float32Array(analyser.fftSize);
  let spoke = false;
  let quietSince = performance.now();
  const started = performance.now();
  let fired = false;
  const tick = setInterval(() => {
    analyser.getFloatTimeDomainData(buf);
    let sum = 0;
    for (const v of buf) sum += v * v;
    const rms = Math.sqrt(sum / buf.length);
    const now = performance.now();
    if (rms > threshold) { spoke = true; quietSince = now; }
    const silentLongEnough = spoke && now - quietSince > silenceMs;
    if (!fired && (silentLongEnough || now - started > maxMs)) { fired = true; onAutoStop(); }
  }, 100);

  const cleanup = () => {
    clearInterval(tick);
    stream.getTracks().forEach((t) => t.stop());
    ctx.close().catch(() => {});
  };

  rec.start(250);
  return {
    stop: () => new Promise<Blob>((resolve) => {
      rec.onstop = () => { cleanup(); resolve(new Blob(chunks, { type: rec.mimeType || "audio/webm" })); };
      if (rec.state !== "inactive") rec.stop(); else { cleanup(); resolve(new Blob(chunks)); }
    }),
    cancel: () => { if (rec.state !== "inactive") rec.stop(); cleanup(); },
    heardSpeech: () => spoke,
  };
}

/** Проиграть звук и дождаться конца. */
export function play(audio: Blob): Promise<void> {
  return new Promise((resolve) => {
    const url = URL.createObjectURL(audio);
    const a = new Audio(url);
    const done = () => { URL.revokeObjectURL(url); resolve(); };
    a.onended = done;
    a.onerror = done;
    a.play().catch(done);
  });
}
